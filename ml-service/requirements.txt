# PS4 detection microservice — pinned dependencies.
# Install into a venv: pip install -r requirements.txt
# CPU-only PyTorch/torchvision wheels — install with the extra index below so
# you get the +cpu build (much smaller download, no CUDA dependency):
#   pip install -r requirements.txt --extra-index-url https://download.pytorch.org/whl/cpu

# --- web service ---
fastapi==0.141.1
uvicorn[standard]==0.52.4
python-multipart==0.0.32
pydantic==2.13.5

# --- model + inference ---
# The "+cpu" local version tag is required, not cosmetic: it pins pip to the
# CPU-only wheel on the extra index below. Without it, pip may resolve the
# default (CUDA-bundled) PyPI build instead, and a torch/torchvision pair
# built for different backends will fail at import time with
# "RuntimeError: operator torchvision::nms does not exist" (hit during setup).
torch==2.13.0+cpu
torchvision==0.28.0+cpu
facenet-pytorch==2.6.0
# Was pinned to 1.29.0 until transformers (added below for audio detection)
# was installed — transformers 4.x requires huggingface_hub<1.0, so pip
# resolved down to 0.36.2. Re-verified hf_hub_download (used by the image
# detector) still imports and works under 0.36.2 before pinning it here.
huggingface_hub==0.36.2
safetensors==0.8.0

# --- Grad-CAM ---

# --- image / io ---

# --- audio detection ---
# Pinned to the 4.x line, not the newer 5.x major (already on PyPI at time
# of writing) — 5.x is a fresh major bump with less field experience behind
# it; 4.x's AutoModelForAudioClassification/AutoFeatureExtractor interface
# is the one actually verified working here. Also the same dependency
# general_ai_detector.py and sdxl_detector.py load their image classifiers
# through (AutoImageProcessor/AutoModelForImageClassification) — no
# separate pin needed for either.
transformers==4.57.6
# Downloads and caches a real static ffmpeg binary on first use — the
# Python-side equivalent of the Next.js side's @ffmpeg-installer/ffmpeg,
# so this service can decode arbitrary audio formats (mp3, ogg, ...) to
# the 16kHz mono waveform the model expects without depending on a system
# ffmpeg install.
imageio-ffmpeg==0.6.0
